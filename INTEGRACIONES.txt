# app/firebase_handler.py
def update_factura_estado(self, factura_id: str, estado: str):
    """
    Update invoice status
    Args:
        factura_id: Document ID
        estado: 'pendiente' | 'revisada' | 'exportada'
    """
    self.db.collection('ocr_invoices').document(factura_id).update({
        'estado': estado,
        'updated_at': datetime.now()
    })

6.4 Marcar múltiples como exportadas:

Python
def mark_facturas_exportadas(self, factura_ids: List[str]):
    """Mark multiple invoices as exported"""
    batch = self.db.batch()
    for factura_id in factura_ids:
        ref = self.db.collection('ocr_invoices').document(factura_id)
        batch.update(ref, {
            'estado': 'exportada',
            'exportada': True,
            'exported_at': datetime.now()
        })
    batch.commit()
